defined_text = {
	name = get_omsk_capital_status
	text = {
		trigger = {
			any_country = {
				FROM = {
					controls_state = 571
				}
			}
		}
		localization_key = "our_control"
	}
	text = {
		trigger = {
			any_country = {
				OR = {
					any_country = {
						is_ally_with = YAK
						controls_state = 571
					}
				}
			}
		}
		localization_key = "enemy_control"
	}
	text = {
		trigger = {
			any_country = {
				NOT = {
					is_ally_with = YAK
				}
				controls_state = 571
			}
		}
		localization_key = "friendly_control"
	}
}
